# engine/game.py
from __future__ import annotations
from abc import ABC, abstractmethod
from typing import Hashable, Sequence

State = Hashable
Action = Hashable


class ExtensiveFormGame(ABC):
    """Two-player zero-sum extensive-form game with chance and imperfect info."""

    num_players: int = 2

    @abstractmethod
    def initial_state(self) -> State: ...

    @abstractmethod
    def is_terminal(self, state: State) -> bool: ...

    @abstractmethod
    def is_chance(self, state: State) -> bool: ...

    @abstractmethod
    def current_player(self, state: State) -> int:
        """0 or 1. Only valid for non-terminal, non-chance states."""

    @abstractmethod
    def legal_actions(self, state: State) -> Sequence[Action]: ...

    @abstractmethod
    def chance_outcomes(self, state: State) -> Sequence[tuple[Action, float]]:
        """(action, probability) pairs summing to 1. Only for chance states."""

    @abstractmethod
    def next_state(self, state: State, action: Action) -> State: ...

    @abstractmethod
    def infoset_key(self, state: State) -> str:
        """Stable key identifying the acting player's information set."""

    @abstractmethod
    def terminal_utility(self, state: State, player: int) -> float:
        """Utility for `player` at a terminal state (zero-sum)."""
```

# engine/games/kuhn.py
from __future__ import annotations
from itertools import permutations
from engine.game import ExtensiveFormGame

# State = ("chance",) | (cards, history)  where cards=(c0,c1), history=str of actions
# Actions: "c" = check/call, "b" = bet, "f" = fold (we model check/bet/call/fold via c/b)
# History grammar (acting player alternates starting P0):
#   ""      P0 to act: c or b
#   "c"     P1 to act: c (->showdown) or b
#   "b"     P1 to act: c (call->showdown) or f-as-c? we use c=call, and "p" pass
# We use a simple alphabet: "p" pass(check/fold), "b" bet/call.

class KuhnPoker(ExtensiveFormGame):
    num_players = 2

    def initial_state(self):
        return ("chance",)

    def is_terminal(self, state) -> bool:
        if state[0] == "chance":
            return False
        _, h = state
        return h in ("pp", "bp", "bb", "pbp", "pbb")

    def is_chance(self, state) -> bool:
        return state[0] == "chance"

    def current_player(self, state) -> int:
        _, h = state
        return len(h) % 2

    def legal_actions(self, state):
        return ("p", "b")

    def chance_outcomes(self, state):
        deals = list(permutations((0, 1, 2), 2))  # 6 ordered deals
        p = 1.0 / len(deals)
        return [(d, p) for d in deals]

    def next_state(self, state, action):
        if state[0] == "chance":
            return (action, "")  # action is the (c0,c1) tuple
        cards, h = state
        return (cards, h + action)

    def infoset_key(self, state) -> str:
        cards, h = state
        return f"{cards[self.current_player(state)]}:{h}"

    def terminal_utility(self, state, player) -> float:
        cards, h = state
        c0, c1 = cards
        # pot contributions beyond the ante
        if h in ("pp",):
            payoff0 = 1 if c0 > c1 else -1            # showdown, pot=2 antes
        elif h == "bp":
            payoff0 = 1                                # P1 folded to bet
        elif h == "pbp":
            payoff0 = -1                               # P0 folded to bet
        elif h in ("bb", "pbb"):
            payoff0 = 2 if c0 > c1 else -2             # showdown, pot includes bet
        else:
            raise ValueError(f"not terminal: {h}")
        return payoff0 if player == 0 else -payoff0
```

# engine/cfr.py
from __future__ import annotations
import numpy as np
from engine.game import ExtensiveFormGame


class _Node:
    __slots__ = ("actions", "regret_sum", "strategy_sum")

    def __init__(self, actions):
        self.actions = list(actions)
        n = len(self.actions)
        self.regret_sum = np.zeros(n)
        self.strategy_sum = np.zeros(n)

    def strategy(self):
        pos = np.maximum(self.regret_sum, 0.0)
        s = pos.sum()
        if s > 0:
            return pos / s
        return np.full(len(self.actions), 1.0 / len(self.actions))


class CFRSolver:
    """CFR family solver. variant in {"cfr", "cfr_plus", "dcfr"}."""

    def __init__(self, game: ExtensiveFormGame, variant: str = "cfr_plus"):
        self.game = game
        self.variant = variant
        self.nodes: dict[str, _Node] = {}
        self._iter = 0
        self._dcfr = (1.5, 0.0, 2.0)  # alpha, beta, gamma

    def _node(self, key, actions) -> _Node:
        node = self.nodes.get(key)
        if node is None:
            node = _Node(actions)
            self.nodes[key] = node
        return node

    def run(self, iterations: int):
        g = self.game
        for _ in range(iterations):
            self._iter += 1
            for player in range(g.num_players):
                self._walk(g.initial_state(), player, 1.0, 1.0)
            self._apply_discount()

    def _apply_discount(self):
        if self.variant != "dcfr":
            return
        a, b, gamma = self._dcfr
        t = self._iter
        pos_d = (t**a) / (t**a + 1)
        neg_d = (t**b) / (t**b + 1)
        strat_d = (t / (t + 1)) ** gamma
        for node in self.nodes.values():
            r = node.regret_sum
            node.regret_sum = np.where(r > 0, r * pos_d, r * neg_d)
            node.strategy_sum *= strat_d

    def _walk(self, state, player, reach_self, reach_opp):
        g = self.game
        if g.is_terminal(state):
            return g.terminal_utility(state, player)
        if g.is_chance(state):
            ev = 0.0
            for action, prob in g.chance_outcomes(state):
                ev += prob * self._walk(g.next_state(state, action), player,
                                        reach_self, reach_opp * prob)
            return ev

        cur = g.current_player(state)
        key = g.infoset_key(state)
        actions = g.legal_actions(state)
        node = self._node(key, actions)
        strategy = node.strategy()

        util = np.zeros(len(actions))
        node_util = 0.0
        for i, action in enumerate(actions):
            nxt = g.next_state(state, action)
            if cur == player:
                util[i] = self._walk(nxt, player, reach_self * strategy[i], reach_opp)
            else:
                util[i] = self._walk(nxt, player, reach_self, reach_opp * strategy[i])
            node_util += strategy[i] * util[i]

        if cur == player:
            regret = util - node_util
            node.regret_sum += reach_opp * regret
            if self.variant == "cfr_plus":
                np.maximum(node.regret_sum, 0.0, out=node.regret_sum)
            weight = reach_self
            if self.variant == "cfr_plus":
                weight = reach_self * self._iter  # linear averaging
            node.strategy_sum += weight * strategy
        return node_util

    def average_strategy(self) -> dict[str, dict]:
        out = {}
        for key, node in self.nodes.items():
            s = node.strategy_sum.sum()
            if s > 0:
                probs = node.strategy_sum / s
            else:
                probs = np.full(len(node.actions), 1.0 / len(node.actions))
            out[key] = {a: float(p) for a, p in zip(node.actions, probs)}
        return out

    def current_strategy(self) -> dict[str, dict]:
        return {key: {a: float(p) for a, p in zip(node.actions, node.strategy())}
                for key, node in self.nodes.items()}

    def game_value(self) -> float:
        """Expected value for player 0 under the average strategy."""
        from engine.exploitability import expected_value
        return expected_value(self.game, self.average_strategy(), 0)
```

# engine/exploitability.py
from __future__ import annotations
from engine.game import ExtensiveFormGame


def expected_value(game: ExtensiveFormGame, strategy: dict, player: int) -> float:
    def walk(state) -> float:
        if game.is_terminal(state):
            return game.terminal_utility(state, player)
        if game.is_chance(state):
            return sum(p * walk(game.next_state(state, a))
                       for a, p in game.chance_outcomes(state))
        key = game.infoset_key(state)
        probs = strategy[key]
        return sum(probs[a] * walk(game.next_state(state, a))
                   for a in game.legal_actions(state))
    return walk(game.initial_state())


def _best_response_value(game, strategy, br_player) -> float:
    # Best response for br_player vs fixed `strategy` for the opponent.
    def walk(state, reach_opp) -> float:
        if game.is_terminal(state):
            return game.terminal_utility(state, br_player) * reach_opp
        if game.is_chance(state):
            return sum(walk(game.next_state(state, a), reach_opp * p)
                       for a, p in game.chance_outcomes(state))
        cur = game.current_player(state)
        actions = list(game.legal_actions(state))
        if cur == br_player:
            # group by infoset: maximize over actions across all states sharing infoset.
            # Simplified exact BR via per-state max is valid only with perfect recall &
            # single-state infosets; for Kuhn/Leduc we compute BR over the infoset tree.
            return max(walk(game.next_state(state, a), reach_opp) for a in actions)
        else:
            probs = strategy[game.infoset_key(state)]
            return sum(walk(game.next_state(state, a), reach_opp * probs[a])
                       for a in actions)
    return walk(game.initial_state(), 1.0)


def exploitability(game: ExtensiveFormGame, strategy: dict) -> float:
    """Sum of both players' best-response gains. 0 == exact Nash."""
    br0 = _best_response_value(game, strategy, 0)
    br1 = _best_response_value(game, strategy, 1)
    return br0 + br1
```

> NOTE during execution: the naive per-state `max` BR above is only correct when
> each infoset maps to a single decision state. Kuhn satisfies this. For Leduc,
> replace with an infoset-grouped BR (accumulate counterfactual values per
> infoset key, then choose the max-value action per infoset). The Leduc task
> includes the corrected version.

```python
def _best_response_value(game, strategy, br_player) -> float:
    # Pass 1: accumulate counterfactual value of each (infoset, action) and the
    # reach-weighted value, then choose the best action per infoset.
    infoset_action_value: dict = {}
    infoset_actions: dict = {}

    def walk(state, reach_opp) -> float:
        if game.is_terminal(state):
            return game.terminal_utility(state, br_player) * reach_opp
        if game.is_chance(state):
            return sum(walk(game.next_state(state, a), reach_opp * p)
                       for a, p in game.chance_outcomes(state))
        cur = game.current_player(state)
        actions = list(game.legal_actions(state))
        if cur != br_player:
            probs = strategy[game.infoset_key(state)]
            return sum(walk(game.next_state(state, a), reach_opp * probs[a])
                       for a in actions)
        key = game.infoset_key(state)
        infoset_actions[key] = actions
        vals = {a: walk(game.next_state(state, a), reach_opp) for a in actions}
        acc = infoset_action_value.setdefault(key, {a: 0.0 for a in actions})
        for a in actions:
            acc[a] += vals[a]
        # provisional: return best; corrected by second pass below
        return max(vals.values())
    # For trees where each infoset is reached once per traversal path, a single
    # pass with per-infoset action accumulation and a top-down best-action choice
    # is exact. We compute total by re-walking with the chosen best actions.
    walk(game.initial_state(), 1.0)
    best_action = {k: max(v, key=v.get) for k, v in infoset_action_value.items()}

    def walk_fixed(state, reach_opp) -> float:
        if game.is_terminal(state):
            return game.terminal_utility(state, br_player) * reach_opp
        if game.is_chance(state):
            return sum(walk_fixed(game.next_state(state, a), reach_opp * p)
                       for a, p in game.chance_outcomes(state))
        cur = game.current_player(state)
        if cur != br_player:
            probs = strategy[game.infoset_key(state)]
            return sum(walk_fixed(game.next_state(state, a), reach_opp * probs[a])
                       for a in game.legal_actions(state))
        a = best_action[game.infoset_key(state)]
        return walk_fixed(game.next_state(state, a), reach_opp)
    return walk_fixed(game.initial_state(), 1.0)
```
